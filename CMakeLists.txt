cmake_minimum_required(VERSION 3.20)
project(KaneDeFlawless LANGUAGES CXX)

add_library(KaneDeFlawless SHARED src/KaneDeFlawless.cpp)
target_compile_features(KaneDeFlawless PRIVATE cxx_std_17)
target_compile_definitions(KaneDeFlawless PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if (MINGW)
    target_compile_options(KaneDeFlawless PRIVATE -m32)
    target_link_options(KaneDeFlawless PRIVATE -m32)
endif()
set_target_properties(KaneDeFlawless PROPERTIES PREFIX "" SUFFIX ".asi")
